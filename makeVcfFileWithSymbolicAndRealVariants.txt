

## Convert gap file from csv to tsv.
cat sample.gaps.csv | tr , "\t" > sample.gaps.tsv

## Extract targets with no coverage from the file PerTargetMetrics.txt.
awk '{if ($7 == 0) print $_}' sample.PerTargetMetrics.txt > sample.PerTargetMetrics0.txt

## Find intersecting regions in gaps.tsv and PerTargetMetrics.txt files.
bedtools intersect -a  sample.gaps.tsv  -b sample.PerTargetMetrics0.txt  > sample.intersections.PerTargetMetrics0.txt

## Make list of intersections to remove from gaps.tsv file.
awk '{print "/"$7"/d"}' sample.intersections.PerTargetMetrics0.txt > sample.intersectionsToDeleteFromGap_tsv.txt

## Remove intersections from the gaps.tsv file.
sed -f sample.intersectionsToDeleteFromGap_tsv.txt sample.gaps.tsv > sample.almostCleanGaps.tsv.txt

## Concatenate gaps and targets with no coverage.
cut -f1-3 sample.almostCleanGaps.tsv.txt sample.PerTargetMetrics0.txt | sed -e '/#/d; /chrM/d; /chrY/d; s/chr//' | bedtools sort > sample.cleanGapsPlusTargets.bed

## To begin constructing the vcf format for the gap or target with no coverage, get the position 5' (immediately to the left) of the start coordinate of the gap or target with no coverage.
awk '{print $1":"$2-1"-"$3}' sample.cleanGapsPlusTargets.bed > sample.GapsPlusTargetsinVcfFormat.txt

## From the human reference genome, obtain the base corresponding to the position captured above (the coordinate immediately to the left of the gap or target region wit no coverage).
samtools faidx Homo_sapiens.GRCh37.75.dna.primary_assembly.fa -r sample.GapsPlusTargetsinVcfFormat.txt  -o sample.GapsPlusTargetsinVcfFormat.fa

## Start constructing the vcf row for each symbolic variant.
## For a homozygote symbolic variant:
sed -e '/^>/N; s/\n/ /' -e 's/>/>chr/; s/ /\t/; s/:/\t/; s/-/\t/' | grep '>' | sed -e 's/>//' | awk 'BEGIN{FS=OFS="\t"}{print $1,$2,".",$4,"<DEL>","999","PASS","END="$3";SVTYPE=DEL;SVLEN=-"$3-$2"\tGT:FT:GQ:PL:PR:SR\t1/1:PASS:999:999,0,999:42,14:262,39"}' > sample.almostSymbolic.tsv

## Get the base to be used for the REF field in the vcf file.
cut -f4 sample.almostSymbolic.tsv | cut -c1 > sample.1bp.txt

## Continue formatting the rows with the symbolic variants.
paste sample.1bp.txt sample.almostSymbolic.tsv | awk '{FS=OFS="\t"}{print $2,$3,$4,$1,$6,$7,$8,$9,$10,$11}' > sample.onlySymbolicVars.tsv

## Get the metadata from an SV file.
bcftools view -h sample.SV.vcf.gz > sample.SVmetadata.txt

## Finish writing the vcf file containing the symbolic variants.
cat sample.SVmetadata.txt sample.onlySymbolicVars.tsv > sample.preFinalSymbolicVars.vcf

## Sort and compress the vcf file with the symbolic variants.
bcftools sort -O z -o sample.finalSymbolicVars0xGT11.vcf.gz sample.preFinalSymbolicVars.vcf

## Index the file.
bcftools index -c -f sample.finalSymbolicVars0xGT11.vcf.gz

## Concatenate the real and symbolic variants.
bcftools concat -a sample.vcf.gz sample.finalSymbolicVars0xGT11.vcf.gz | bcftools sort -O z -o sample.Vars0xGT11.vcf.gz

## Index the vcf file having real and symbolic variants.
bcftools index -c -f sample.Vars0xGT11.vcf.gz



## For a heterozyote symbolic variant:
sed -e '/^>/N; s/\n/ /' -e 's/>/>chr/; s/ /\t/; s/:/\t/; s/-/\t/' | grep '>' | sed -e 's/>//' | awk 'BEGIN{FS=OFS="\t"}{print $1,$2,".",$4,"<DEL>","999","PASS","END="$3";SVTYPE=DEL;SVLEN=-"$3-$2"\tGT:FT:GQ:PL:PR:SR\t0/1:PASS:999:999,0,999:42,14:262,39"}' > sample.almostSymbolic.tsv

## Get the base to be used for the REF field in the vcf file.
cut -f4 sample.almostSymbolic.tsv | cut -c1 > sample.1bp.txt

## Continue formatting the rows with the symbolic variants.
